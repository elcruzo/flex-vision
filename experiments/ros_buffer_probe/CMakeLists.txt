cmake_minimum_required(VERSION 3.20)
project(cpg_buffer_probe)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rosidl_buffer REQUIRED)
add_executable(image_probe src/image_probe.cpp)
target_compile_features(image_probe PRIVATE cxx_std_17)
target_compile_options(image_probe PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(image_probe rclcpp sensor_msgs rosidl_buffer)
install(TARGETS image_probe DESTINATION lib/${PROJECT_NAME})
ament_package()
